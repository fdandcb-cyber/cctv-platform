   150→   150→   150→    try {
   151→   151→   151→      const stored = localStorage.getItem("cctv_saved_quotes");

   160→   160→   160→    if (totalCams === 0) {

   162→   162→   162→      return;

   165→   165→   165→    const quote: SavedQuote = {
   166→   166→   166→      id: Date.now().toString(),

   168→   168→   168→      date: new Date().toLocaleDateString("en-IN"),
   169→   169→   169→      data: JSON.stringify({
   170→   170→   170→        propertyType: state.propertyType,
   171→   171→   171→        areaSqft: state.areaSqft,
   172→   172→   172→        cameraSystem: state.cameraSystem,
   173→   173→   173→        cameraTechs: state.cameraTechs,
   174→   174→   174→        cameraSelections: state.cameraSelections,
   175→   175→   175→        recorderUnits: state.recorderUnits,
   176→   176→   176→        powerUnits: state.powerUnits,
   177→   177→   177→        retentionDays: state.retentionDays,

   194→   194→   194→      summary: `${totalCams} cameras | ${state.cameraSystem.toUpperCase()} | ${state.areaSqft} sq ft`,
   195→   195→   195→    };
   196→   196→   196→    const updated = [quote, ...quotes];
   197→   197→   197→    setQuotes(updated);
   198→   198→   198→    localStorage.setItem("cctv_saved_quotes", JSON.stringify(updated));
   199→   199→   199→    toast.success("Quote saved!");
   200→   200→   200→  };
   201→   201→   201→
   202→   202→   202→  const deleteQuote = (id: string) => {
   203→   203→   203→    if (!confirm("Delete this saved quote?")) return;
   204→   204→   204→    const updated = quotes.filter(q => q.id !== id);
   205→   205→   205→    setQuotes(updated);
   206→   206→   206→    localStorage.setItem("cctv_saved_quotes", JSON.stringify(updated));
   207→   207→   207→    toast.success("Quote deleted");
   208→   208→   208→  };
   209→   209→   209→
   210→   210→   210→  const loadQuote = (quote: SavedQuote) => {
   211→   211→   211→    try {
   212→   212→   212→      const data = JSON.parse(quote.data);
   213→   213→   213→      const store = useBuilderStore.getState();
   214→   214→   214→      Object.keys(data).forEach((key) => {
   215→   215→   215→        const actionKey = "set" + key.charAt(0).toUpperCase() + key.slice(1) as keyof typeof store;
   216→   216→   216→        if (typeof store[actionKey] === "function") {
   217→   217→   217→          (store[actionKey] as (v: unknown) => void)(data[key]);
   218→   218→   218→        }
   219→   219→   219→      });
   220→   220→   220→      setLoadedQuote(quote.id);
   221→   221→   221→      toast.success("Quote loaded! Go to Builder tab to view.");

   224→   224→   224→      toast.error("Failed to load quote");

   279→   279→   279→      if (data.ups > 0) lines.push("UPS: " + data.ups);
   280→   280→   280→
   281→   281→   281→      navigator.clipboard.writeText(lines.join("\n")).then(() => toast.success("Quote copied!")).catch(() => toast.error("Copy failed"));
   282→   282→   282→    } catch { toast.error("Failed to read quote data"); }
   283→   283→   283→  };
   284→   284→   284→
   285→   285→   285→  return (

   295→   295→   295→      </div>
   296→   296→   296→
   297→   297→   297→      {quotes.length === 0 ? (
   298→   298→   298→        <div className="text-center py-12 bg-muted/30 rounded-xl border border-dashed">
