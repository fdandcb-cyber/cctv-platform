    34→    34  function AnimatedCounter({
    35→    35    target,
    36→    36    suffix = "",
    37→    37    prefix = "",
    38→    38    duration = 2000,
    39→    39  }: {
    40→    40    target: number;
    41→    41    suffix?: string;
    42→    42    prefix?: string;
    43→    43    duration?: number;
    44→    44  }) {
    45→    45    const ref = useRef<HTMLSpanElement>(null);
    46→    46    const inView = useInView(ref, { once: true, margin: "-50px" });
    47→    47    const [count, setCount] = useState(0);
    48→    48  
    49→    49    useEffect(() => {
    50→    50      if (!inView) return;
    51→    51      let start = 0;
    52→    52      const step = Math.ceil(target / (duration / 16));
    53→    53      const timer = setInterval(() => {
    54→    54        start += step;
    55→    55        if (start >= target) {
    56→    56          setCount(target);
    57→    57          clearInterval(timer);
    58→    58        } else {
    59→    59          setCount(start);
    60→    60        }
    61→    61      }, 16);
    62→    62      return () => clearInterval(timer);
    63→    63    }, [inView, target, duration]);
    64→    64  
    65→    65    return (
    66→    66      <span ref={ref}>
    67→    67        {prefix}{count.toLocaleString("en-IN")}{suffix}
    68→    68      </span>
    69→    69    );
    70→    70  }
